2) Nombre del visitante que más visitas ha hecho a cada museo.

SELECT V.Cod_M, V.DNI_V,D.Nombre_V, COUNT(*)
FROM Datos_Visitantes D, Visitas V
WHERE D.DNI_V = V.DNI_V
GROUP BY V.Cod_M, V.DNI_V
HAVING COUNT(*) >= ALL (SELECT COUNT(*) FROM Visitas WHERE Cod_M = V.Cod_M GROUP BY DNI_V);

SELECT Nombre_V
FROM Museos M, Visitantes V, Datos_Visitantes D
WHERE M.Cod_M = V.Cod_M
AND D.DNI_V = V.DNI_V
AND Nombre_M = 'Museo 1'
AND D.DNI_V NOT IN (SELECT DNI_V
                    FROM Museos M1, Visitantes V1
                    WHERE M1.Cod_M = V1.Cod_M
                    AND Nombre_M <> 'Museo 1');

CREATE TABLE VIGILANTES(
  DNI_V VARCHAR(9) PRIMARY KEY,
  Cod_M INTEGER NOT NULL REFERENCES Museos(Cod_M) ON DELETE CASCADE ON UPDATE CASCADE,
  NSS_V INTEGER NOT NULL,
  NControl_V INTEGER NOT NULL,
  NArma_V INTEGER NOT NULL,
  Coordinador_V CHARACTER(9) REFERENCES VIGILANTES(DNI_V) ON DELETE SET NULL ON UPDATE CASCADE);
